Database Schema for Real-time Tracking
sql-- Messages table with full tracking
CREATE TABLE messages (
  id SERIAL PRIMARY KEY,
  whatsapp_message_id VARCHAR(255) UNIQUE,
  campaign_id INTEGER REFERENCES campaigns(id),
  recipient_phone VARCHAR(20),
  template_id INTEGER REFERENCES templates(id),
  status VARCHAR(20), -- queued, sent, delivered, read, failed
  queued_at TIMESTAMP,
  sent_at TIMESTAMP,
  delivered_at TIMESTAMP,
  read_at TIMESTAMP,
  error_code VARCHAR(50),
  error_description TEXT,
  cost DECIMAL(10, 4),
  metadata JSONB
);

-- Index for fast lookups
CREATE INDEX idx_messages_whatsapp_id ON messages(whatsapp_message_id);
CREATE INDEX idx_messages_campaign ON messages(campaign_id);
CREATE INDEX idx_messages_status ON messages(status);

-- Templates with Meta sync data
CREATE TABLE templates (
  id SERIAL PRIMARY KEY,
  meta_template_id VARCHAR(255) UNIQUE,
  name VARCHAR(512),
  category VARCHAR(50),
  language VARCHAR(10),
  status VARCHAR(20), -- PENDING, APPROVED, REJECTED, DISABLED
  quality_score VARCHAR(20), -- GREEN, YELLOW, RED
  rejection_reason TEXT,
  components JSONB,
  last_synced_at TIMESTAMP,
  created_at TIMESTAMP,
  updated_at TIMESTAMP
);

-- Campaign metrics (updated in real-time)
CREATE TABLE campaign_metrics (
  campaign_id INTEGER PRIMARY KEY REFERENCES campaigns(id),
  total_messages INTEGER DEFAULT 0,
  sent_count INTEGER DEFAULT 0,
  delivered_count INTEGER DEFAULT 0,
  read_count INTEGER DEFAULT 0,
  failed_count INTEGER DEFAULT 0,
  total_cost DECIMAL(10, 2) DEFAULT 0,
  last_updated_at TIMESTAMP,
  
  -- Percentages calculated on read
  delivery_rate DECIMAL(5, 2) GENERATED ALWAYS AS 
    (CASE WHEN sent_count > 0 THEN (delivered_count::DECIMAL / sent_count * 100) ELSE 0 END) STORED,
  read_rate DECIMAL(5, 2) GENERATED ALWAYS AS 
    (CASE WHEN delivered_count > 0 THEN (read_count::DECIMAL / delivered_count * 100) ELSE 0 END) STORED
);
